# works around this by exporting:
#   export DYLD_FALLBACK_LIBRARY_PATH=/opt/homebrew/lib:$DYLD_FALLBACK_LIBRARY_PATH
# before invoking WeasyPrint. No need to recreate the venv with Homebrew Python.
